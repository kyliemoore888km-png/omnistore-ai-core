def log_audit(db: Session, user_id: str | None, team_id: str | None, action: str, resource_type: str, resource_id: str | None = None, changes: str | None = None):
    db.add(
        AuditLog(
            id=str(uuid.uuid4()),
            user_id=user_id,
            team_id=team_id,
            action=action,
            resource_type=resource_type,
            resource_id=resource_id,
            changes=changes,
            ip_address="system",
            user_agent="server",
        )
    )
    db.commit()


def get_validated_user(token: str | None, db: Session):
    if not token:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Missing token")
    token_data = verify_token(token)
    if not token_data:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid token")
    user = db.query(User).filter(User.id == token_data.user_id).first()
    if not user:
        raise HTTPException(status_code=status.HTTP_404_NOT_FOUND, detail="User not found")
    return user, token_data

@app.post("/api/auth/signup", response_model=TokenResponse)
async def signup(user_create: UserCreate, db: Session = Depends(get_db)):
    existing = db.query(User).filter(User.email == user_create.email).first()
    if existing:
        raise HTTPException(status_code=400, detail="Email already registered")

    team_id = str(uuid.uuid4())
    user_id = str(uuid.uuid4())

    team = Team(
        id=team_id,
        name=f"{user_create.full_name}'s Team",
        slug=user_create.email.split("@")[0].lower(),
        owner_id=user_id,
        subscription_tier=SubscriptionTier.FREE,
        max_users=5,
        max_stores=2,
    )

    user = User(
        id=user_id,
        email=user_create.email,
        full_name=user_create.full_name,
        hashed_password=hash_password(user_create.password),
        role=UserRole.ADMIN,
        team_id=team_id,
        is_active=True,
        mfa_enabled=False,
    )

    db.add(team)
    db.add(user)
    db.commit()
    db.refresh(user)

    log_audit(db, user.id, team.id, "SIGNUP", "user", user.id)

    access_token = create_access_token({
        "sub": user.email,
        "user_id": user.id,
        "role": user.role.value,
        "team_id": team.id,
    })
    refresh_token = create_refresh_token({"sub": user.email})

    return TokenResponse(
        access_token=access_token,
        refresh_token=refresh_token,
        token_type="bearer",
        user=UserResponse.from_orm(user),
    )


@app.post("/api/auth/login", response_model=TokenResponse)
async def login(login_req: LoginRequest, db: Session = Depends(get_db)):
    user = db.query(User).filter(User.email == login_req.email).first()
    if not user or not verify_password(login_req.password, user.hashed_password):
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid credentials")

    if user.mfa_enabled and not login_req.mfa_code:
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="MFA required")

    access_token = create_access_token({
        "sub": user.email,
        "user_id": user.id,
        "role": user.role.value,
        "team_id": user.team_id,
    })
    refresh_token = create_refresh_token({"sub": user.email})

    log_audit(db, user.id, user.team_id, "LOGIN", "user", user.id)

    return TokenResponse(
        access_token=access_token,
        refresh_token=refresh_token,
        token_type="bearer",
        user=UserResponse.from_orm(user),
    )

@app.post("/api/stores")
async def create_store(store_create: StoreCreate, token: str = Query(default=None), db: Session = Depends(get_db)):
    user, token_data = get_validated_user(token, db)
    if user.role not in [UserRole.ADMIN, UserRole.MANAGER]:
        raise HTTPException(status_code=403, detail="Insufficient permissions")

    team = db.query(Team).filter(Team.id == user.team_id).first()
    if not team:
        raise HTTPException(status_code=404, detail="Team not found")

    existing_store_count = db.query(Store).filter(Store.team_id == team.id).count()
    if existing_store_count >= team.max_stores:
        raise HTTPException(status_code=400, detail="Store limit reached")

    store = Store(
        id=str(uuid.uuid4()),
        name=store_create.name,
        team_id=team.id,
        platform=store_create.platform,
        api_key=store_create.api_key,
        api_secret=store_create.api_secret,
        is_active=True,
    )
    db.add(store)
    db.commit()
    log_audit(db, user.id, team.id, "CREATE_STORE", "store", store.id)
    return StoreResponse.from_orm(store)

@app.post("/api/orders")
async def create_order(order_create: OrderCreate, store_id: str, token: str = Query(default=None), db: Session = Depends(get_db)):
    user, _ = get_validated_user(token, db)
    store = db.query(Store).filter(Store.id == store_id, Store.team_id == user.team_id).first()
    if not store:
        raise HTTPException(status_code=404, detail="Store not found")

    order = Order(
        id=str(uuid.uuid4()),
        order_number=f"ORD-{datetime.utcnow().strftime('%Y%m%d')}-{str(uuid.uuid4())[:8]}",
        user_id=user.id,
        store_id=store_id,
        status=OrderStatus.PENDING,
        total_amount=order_create.total_amount,
        customer_email=order_create.customer_email,
        customer_name=order_create.customer_name,
        shipping_address=order_create.shipping_address,
        notes="",
    )
    db.add(order)
    db.commit()
    db.refresh(order)

    for item in order_create.items:
        db.add(
            OrderItem(
                id=str(uuid.uuid4()),
                order_id=order.id,
                product_id=item.product_id,
                quantity=item.quantity,
                price=item.price,
            )
        )
    db.commit()

    log_audit(db, user.id, user.team_id, "CREATE_ORDER", "order", order.id)
    return OrderResponse.from_orm(order)

@app.get("/api/analytics/dashboard")
async def dashboard(token: str = Query(default=None), db: Session = Depends(get_db)):
    user, _ = get_validated_user(token, db)
    total_orders = db.query(Order).join(Store).filter(Store.team_id == user.team_id).count()
    total_revenue = db.query(func.sum(Order.total_amount)).join(Store).filter(Store.team_id == user.team_id).scalar() or 0
    active_products = db.query(Product).join(Store).filter(Store.team_id == user.team_id, Product.is_active == True).count()
    low_stock = db.query(Product).join(Store).filter(Store.team_id == user.team_id, Product.stock_quantity <= Product.reorder_level).count()

    recent_orders = db.query(Order).join(Store).filter(Store.team_id == user.team_id).order_by(Order.created_at.desc()).limit(5).all()

    return {
        "overview": {
            "total_orders": total_orders,
            "total_revenue": float(total_revenue),
            "active_products": active_products,
            "low_stock_alerts": low_stock,
        },
        "recent_orders": [
            {
                "id": order.id,
                "order_number": order.order_number,
                "customer_name": order.customer_name,
                "total_amount": float(order.total_amount),
                "status": order.status.value,
            }
            for order in recent_orders
        ],
    }


@app.get("/api/analytics/revenue")
async def revenue_analytics(token: str = Query(default=None), days: int = 30, db: Session = Depends(get_db)):
    user, _ = get_validated_user(token, db)
    start = datetime.utcnow() - timedelta(days=days)

    rows = (
        db.query(func.date(Order.created_at).label("date"), func.sum(Order.total_amount).label("revenue"), func.count(Order.id).label("orders"))
        .join(Store)
        .filter(Store.team_id == user.team_id, Order.created_at >= start)
        .group_by(func.date(Order.created_at))
        .all()
    )

    return {
        "period_days": days,
        "data": [{"date": str(r[0]), "revenue": float(r[1]), "orders": r[2]} for r in rows],
    }

@app.post("/api/subscriptions/upgrade")
async def upgrade_subscription(tier: SubscriptionTier, token: str = Query(default=None), db: Session = Depends(get_db)):
    user, _ = get_validated_user(token, db)
    if user.role != UserRole.ADMIN:
        raise HTTPException(status_code=403, detail="Admin required")

    team = db.query(Team).filter(Team.id == user.team_id).first()
    if not team:
        raise HTTPException(status_code=404, detail="Team not found")

    limits = {
        SubscriptionTier.FREE: {"max_users": 5, "max_stores": 2},
        SubscriptionTier.STARTER: {"max_users": 10, "max_stores": 5},
        SubscriptionTier.PROFESSIONAL: {"max_users": 50, "max_stores": 20},
        SubscriptionTier.ENTERPRISE: {"max_users": 500, "max_stores": 200},
    }

    subscription = db.query(Subscription).filter(Subscription.team_id == team.id).first()
    if not subscription:
        subscription = Subscription(id=str(uuid.uuid4()), team_id=team.id, tier=tier, status="active", monthly_cost=0)
        db.add(subscription)
    else:
        subscription.tier = tier

    team.subscription_tier = tier
    team.max_users = limits[tier]["max_users"]
    team.max_stores = limits[tier]["max_stores"]
    db.commit()

    log_audit(db, user.id, team.id, "UPGRADE_SUBSCRIPTION", "subscription", subscription.id)
    return {"message": f"Subscription upgraded to {tier.value}"}
